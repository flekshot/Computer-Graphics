cmake_minimum_required(VERSION 3.24)
project(SolarSystemExplorer VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SSE_BUILD_TESTS "Build CPU and OpenGL integration tests" ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/glfw EXCLUDE_FROM_ALL)
find_package(OpenGL REQUIRED)
add_library(glm INTERFACE)
target_include_directories(glm SYSTEM INTERFACE third_party/glm)
add_library(glew STATIC third_party/glew/src/glew.c)
target_include_directories(glew SYSTEM PUBLIC third_party/glew/include)
target_compile_definitions(glew PUBLIC GLEW_STATIC GLEW_NO_GLU)
target_link_libraries(glew PUBLIC OpenGL::GL)
add_library(imgui STATIC
    third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
    third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
    third_party/imgui/backends/imgui_impl_glfw.cpp
    third_party/imgui/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui SYSTEM PUBLIC third_party/imgui third_party/imgui/backends)
target_link_libraries(imgui PUBLIC glfw OpenGL::GL)

add_library(sse_core STATIC src/Planet.cpp src/SolarSystem.cpp src/Camera.cpp src/Geometry.cpp
    src/ExperimentIO.cpp src/EclipseModel.cpp)
target_include_directories(sse_core PUBLIC include)
target_link_libraries(sse_core PUBLIC glm)
add_executable(SolarSystemExplorer src/main.cpp src/Application.cpp src/Shader.cpp src/Mesh.cpp src/Texture.cpp
    src/Renderer.cpp src/UIManager.cpp src/LabUI.cpp src/EclipseRenderer.cpp src/ExperimentDialogs.cpp)
target_include_directories(SolarSystemExplorer PRIVATE include third_party/stb)
target_compile_definitions(SolarSystemExplorer PRIVATE
    GLFW_INCLUDE_NONE NOMINMAX WIN32_LEAN_AND_MEAN
    SSE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(SolarSystemExplorer PRIVATE glew glfw glm imgui sse_core)
if(WIN32)
    target_link_libraries(SolarSystemExplorer PRIVATE comdlg32)
endif()
if(MSVC)
    target_compile_options(SolarSystemExplorer PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(SolarSystemExplorer PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_command(TARGET SolarSystemExplorer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:SolarSystemExplorer>/shaders"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:SolarSystemExplorer>/assets")
set_property(TARGET SolarSystemExplorer PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:SolarSystemExplorer>")
install(TARGETS SolarSystemExplorer RUNTIME DESTINATION .)
install(DIRECTORY shaders assets licenses DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(DIRECTORY examples DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
include(CTest)
if(SSE_BUILD_TESTS)
    target_compile_definitions(SolarSystemExplorer PRIVATE SSE_WITH_TESTS)
    target_include_directories(SolarSystemExplorer PRIVATE tests)
    target_sources(SolarSystemExplorer PRIVATE tests/SmokeScenario.cpp tests/LabSmokeScenario.cpp)
    add_executable(CoreTests tests/CoreTests.cpp)
    target_link_libraries(CoreTests PRIVATE sse_core)
    add_test(NAME core_behaviors COMMAND CoreTests)
    add_executable(LabTests tests/LabTests.cpp)
    target_link_libraries(LabTests PRIVATE sse_core)
    add_test(NAME orbit_and_persistence COMMAND LabTests)
    add_executable(EclipseTests tests/EclipseTests.cpp)
    target_link_libraries(EclipseTests PRIVATE sse_core)
    add_test(NAME eclipse_geometry COMMAND EclipseTests)
    add_test(NAME opengl_context COMMAND SolarSystemExplorer --frames 3 --hidden)
    add_test(NAME graphics_smoke COMMAND SolarSystemExplorer --smoke-test --hidden)
    add_test(NAME laboratory_smoke COMMAND SolarSystemExplorer --lab-smoke-test --hidden)
    add_test(NAME missing_textures COMMAND SolarSystemExplorer --frames 5 --hidden --fallback-textures)
    add_test(NAME no_bloom COMMAND SolarSystemExplorer --frames 5 --hidden --no-bloom)
    add_test(NAME invalid_cli COMMAND SolarSystemExplorer --speed nan)
    if(WIN32)
        add_test(NAME resource_recovery COMMAND ${CMAKE_COMMAND}
            "-DEXE=$<TARGET_FILE:SolarSystemExplorer>" "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}"
            "-DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/resource-tests" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/ResourceTests.cmake")
        set_tests_properties(resource_recovery PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
    endif()
    set_tests_properties(invalid_cli PROPERTIES WILL_FAIL TRUE)
    set_tests_properties(opengl_context graphics_smoke laboratory_smoke missing_textures no_bloom PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
endif()
